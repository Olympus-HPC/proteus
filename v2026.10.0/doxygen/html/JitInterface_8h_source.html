105template <uint64_t FunctorID, typename Lambda> struct LambdaFunctorWrapper {
106 using LambdaType = Lambda;
107 static constexpr std::uint64_t functor_id = FunctorID;
108 LambdaType lambda;
109
110 template <typename... Args>
111 PROTEUS_HOST_DEVICE __attribute__((annotate("jit")))
112 __attribute__((annotate("proteus.wrapper_call", functor_id))) decltype(auto)
113 operator()(Args &&...args) const noexcept {
114 return lambda(std::forward<Args>(args)...);
115 }
116};

118template <typename... T> struct is_lambda_functor_wrapper : std::false_type {};

120template <uint64_t FunctorID, typename Lambda>
121struct is_lambda_functor_wrapper<LambdaFunctorWrapper<FunctorID, Lambda>>
122 : std::true_type {};

124template <std::uint64_t FunctorId, typename L>
125PROTEUS_HOST_DEVICE inline auto tag_functor(L &&lambda) {
126 return LambdaFunctorWrapper<FunctorId, std::decay_t<L>>{
127 std::forward<L>(lambda)};

130template <uint64_t ID, typename T>
131[[nodiscard]] static __attribute__((noinline))
132__attribute__((annotate("proteus.register_call_impl", ID))) auto
133__register_lambda_impl(T &&t) noexcept {
134 static_assert(!detail::is_lambda_functor_wrapper<std::decay_t<T>>::value);
135 // Force LLVM to generate an AllocaInst of the underlying Clang--generated
136 // anonymous class for T. We remove this after recording the demangled
137 // lambda name.
138 using LambdaType = std::decay_t<T>;
139 LambdaType local = t;
140 __proteus_take_address(&local);
141 auto result = tag_functor<ID>(std::forward<T>(t));
142 return result;
143}
