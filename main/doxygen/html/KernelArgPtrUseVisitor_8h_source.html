61inline std::optional<MemoryLocation>
62getTrackedPointerLocation(const DataLayout &DL, Value *Ptr) {
63 if (!Ptr || !Ptr->getType()->isPointerTy())
64 return std::nullopt;
65
66 Type *PointeeTy = nullptr;
67 if (auto *AI = dyn_cast<AllocaInst>(Ptr))
68 PointeeTy = AI->getAllocatedType();
69
70 if (!PointeeTy || !PointeeTy->isSized())
71 return MemoryLocation::getBeforeOrAfter(Ptr);
72
73 return MemoryLocation(Ptr,
74 LocationSize::precise(DL.getTypeStoreSize(PointeeTy)));
75}

92 FunctionAnalyses &getAnalyses(Function &F) {
93 auto [It, Inserted] = Cache.try_emplace(&F);
94 if (Inserted) {
95 It->second.TLII = std::make_unique<TargetLibraryInfoImpl>(
96 Triple(F.getParent()->getTargetTriple()));
97 It->second.TLI = std::make_unique<TargetLibraryInfo>(*It->second.TLII);

99 It->second.DT = std::make_unique<DominatorTree>(F);
100 It->second.AA = std::make_unique<AAResults>(*It->second.TLI);
101 It->second.BAA = std::make_unique<BasicAAResult>(
102 DL, F, *It->second.TLI, *It->second.AC, It->second.DT.get());
103 It->second.AA->addAAResult(*It->second.BAA);

110 std::optional<LambdaPtrUseAnalysis> resolveMemoryAccess(MemoryAccess *MA,
111 int64_t Offset) {
112 if (!MA)
113 return std::nullopt;
114
115 if (auto *MP = dyn_cast<MemoryPhi>(MA)) {
116 std::optional<LambdaPtrUseAnalysis> Common;
117 for (Use &Incoming : MP->incoming_values()) {
118 auto *IncomingMA = dyn_cast<MemoryAccess>(Incoming.get());
119 if (!IncomingMA)
120 return std::nullopt;
121 auto Res = resolveMemoryAccess(IncomingMA, Offset);
122 if (!Res)
123 return std::nullopt;
124 if (!Common) {
125 Common = Res;
126 continue;
127 }
128 if (Common->DominatingWrite != Res->DominatingWrite ||
129 Common->Offset != Res->Offset ||
130 Common->ChangedRCLayout != Res->ChangedRCLayout)
131 return std::nullopt;
132 }
133 return Common;
134 }
135
136 auto *MD = dyn_cast<MemoryDef>(MA);
137 if (!MD)
138 return std::nullopt;
139
140 auto *MemI = dyn_cast_or_null<Instruction>(MD->getMemoryInst());
141 if (!MemI)
142 return std::nullopt;
143
144 if (auto *SI = dyn_cast<StoreInst>(MemI)) {
145 Value *Stored = SI->getValueOperand();
146 if (!Stored->getType()->isPointerTy())
147 return std::nullopt;
148 return LambdaPtrUseAnalysis{Stored, Offset, std::nullopt};
149 }
150
151 auto *MT = dyn_cast<MemTransferInst>(MemI);
152 if (!MT)
153 return std::nullopt;
154
155 int64_t DstOff = 0, SrcOff = 0;
156 Value *DstBase =
157 GetPointerBaseWithConstantOffset(MT->getRawDest(), DstOff, DL);
158 Value *SrcBase =
159 GetPointerBaseWithConstantOffset(MT->getRawSource(), SrcOff, DL);
160 if (!DstBase || !SrcBase)
161 return std::nullopt;
162
163 return LambdaPtrUseAnalysis{SrcBase, Offset - DstOff + SrcOff,
164 std::nullopt};
165 }

170 std::optional<LambdaPtrUseAnalysis> resolve(Value *TrackedPtr,
171 Instruction *UseI) {
172 auto Loc = getTrackedPointerLocation(DL, TrackedPtr);
173 if (!Loc)
174 return std::nullopt;
175
176 Function &F = *UseI->getFunction();
177 auto &Analyses = getAnalyses(F);
178 auto *MA = Analyses.MSSA->getMemoryAccess(UseI);
179 if (!MA)
180 return std::nullopt;
181
182 auto *Walker = Analyses.MSSA->getWalker();
183 auto *Clobber = Walker->getClobberingMemoryAccess(MA, *Loc);
184 return resolveMemoryAccess(Clobber, /*Offset=*/0);
185 }

355inline std::optional<LambdaPtrUseAnalysis>
356getDominatingUse(const DataLayout &DL, Value *ValueNeedingAnalysis,
357 Value *SeenUse) {
358 DEBUG(Logger::logs("proteus-pass") << "Beginning PtrUse analysis " << "\n");
359
360 if (auto *I = dyn_cast<Instruction>(SeenUse)) {
361 FunctionMemorySSAResolver Resolver(DL);
362 if (auto Res = Resolver.resolve(ValueNeedingAnalysis, I)) {
363 DEBUG(Logger::logs("proteus-pass")
364 << " [PTR use analysis]: MemorySSA resolved "
365 << *ValueNeedingAnalysis << " at use " << *SeenUse << "\n");
366 return Res;
367 }
368 }
369
370 LambdaInstUseVisitor Visitor(ValueNeedingAnalysis, SeenUse, DL);
371 // Analysis loop
372 while (!Visitor.empty() && !Visitor.success() && !Visitor.failed()) {
373 auto *V = Visitor.back();
374 Visitor.popBack();
375 // Prevent loops/infinite recursion
376 if (Visitor.seen(V))
377 continue;
378 Visitor.markAsSeen(V);
379 DEBUG(Logger::logs("proteus-pass")
380 << " [PTR use analysis]: Visiting ptr use " << *V << "\n");
381 // Analyze the instruction
382 if (auto *I = dyn_cast<Instruction>(V))
383 Visitor.visit(*I);
384 else
385 continue;
386 }
387 if (!Visitor.success() || Visitor.failed()) {
388 DEBUG(
389 Logger::logs("proteus-pass")
390 << " [PTR use analysis] [WARNING]: Dominating use analysis FAILED for "
391 << *ValueNeedingAnalysis << " <-- " << *SeenUse << "\n");
392 return std::nullopt;
393 }
394 LambdaPtrUseAnalysis Info = Visitor.getAnalysisResult();
395 if (!Info.DominatingWrite)
396 return std::nullopt;
397 DEBUG(Logger::logs("proteus-pass")
398 << " [PTR USE ANALYSIS]: Computed offset " << Info.Offset << "\n");
399 return Info;
400
401 return std::nullopt;
402}
