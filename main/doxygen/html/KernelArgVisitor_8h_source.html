44inline std::optional<int64_t>
45getAggregateIndicesOffset(const DataLayout &DL, Type *AggTy,
46 ArrayRef<unsigned> Indices) {
47 int64_t Offset = 0;
48 Type *CurTy = AggTy;
49 for (unsigned Idx : Indices) {
50 if (auto *ST = dyn_cast<StructType>(CurTy)) {
51 if (Idx >= ST->getNumElements())
52 return std::nullopt;
53 const StructLayout *SL = DL.getStructLayout(ST);
54 Offset += static_cast<int64_t>(SL->getElementOffset(Idx));
55 CurTy = ST->getElementType(Idx);
56 continue;
57 }
58
59 if (auto *AT = dyn_cast<ArrayType>(CurTy)) {
60 if (Idx >= AT->getNumElements())
61 return std::nullopt;
62 Offset +=
63 static_cast<int64_t>(DL.getTypeAllocSize(AT->getElementType())) * Idx;
64 CurTy = AT->getElementType();
65 continue;
66 }
67
68 return std::nullopt;
69 }
70 return Offset;
71}

160 inline std::optional<LambdaKernelArgAnalysis>
161 cloneAndAnalyze(Value *Start, Value *MemoryAnalysisPtrUse,
162 int64_t StartOffset) {
163 LambdaArgVisitor Visitor(Start, MemoryAnalysisPtrUse, StartOffset, DL);
164 while (!Visitor.empty() && !Visitor.success() && !Visitor.failed()) {
165 auto [V, AccessedFrom] = Visitor.back();
166 Visitor.MemoryAnalysisPtrUse = AccessedFrom;
167 Visitor.popBack();
168 // Prevent loops/infinite recursion
169 if (Visitor.seen(V))
170 continue;
171 Visitor.markAsSeen(V);
172 // Analyze the instruction
173 if (auto *I = dyn_cast<Instruction>(V))
174 Visitor.visit(*I);
175 else if (auto *A = dyn_cast<Argument>(V))
176 Visitor.visitArgument(*A);
177 else
178 continue;
179 }
180 if (!Visitor.success() || Visitor.failed())
181 return std::nullopt;
182 return Visitor.getKernelArgInfo();
183 }

185 inline std::optional<FunctionAnalysis> analyzeFunction(CallBase &CB,
186 int64_t StartOffset) {
187 FunctionAnalysis Result;
188 auto &F = *CB.getCalledFunction();
189 auto RetInstOpt = getRetInst(F);
190 if (!RetInstOpt)
191 return std::nullopt;
192 LambdaArgVisitor Visitor(RetInstOpt.value()->getReturnValue(),
193 MemoryAnalysisPtrUse, StartOffset, DL);
194 while (!Visitor.empty() && !Visitor.success() && !Visitor.failed()) {
195 auto [V, AccessedFrom] = Visitor.back();
196 DEBUG(Logger::logs("proteus-pass")
197 << "Function analysis visiting " << *V << " with offset "
198 << Visitor.getOffset() << "\n");
199 Visitor.MemoryAnalysisPtrUse = AccessedFrom;
200 Visitor.popBack();
201 // Prevent loops/infinite recursion
202 if (Visitor.seen(V))
203 continue;
204 Visitor.markAsSeen(V);
205 // Analyze the instruction
206 if (auto *I = dyn_cast<Instruction>(V))
207 Visitor.visit(*I);
208 else if (auto *A = dyn_cast<Argument>(V)) {
209 if (A->getParent() == CB.getCalledFunction()) {
210 Result.PtrArgToCB = CB.getArgOperand(A->getArgNo());
211 Result.Offset = Visitor.Offset;
212 Result.ArgIndex = A->getArgNo();
213 DEBUG(Logger::logs("proteus-pass")
214 << "Function analysis found termination case "
215 << *Result.PtrArgToCB << " with offset " << Visitor.getOffset()
216 << "\n");
217 return Result;
218 }
219 Visitor.visitArgument(*A);
220
221 } else
222 continue;
223 }
224
225 return std::nullopt;

255 void visitLoadInst(LoadInst &LI) {
256 // TODO: Delete this? We shouldn't reach this from a use-def
257 if (!LI.getType()->isPointerTy()) {
258 AnalysisFailed = true;
259 return;
260 }
261
262 int64_t LoadOff = 0;
263 Value *LoadBase =
264 GetPointerBaseWithConstantOffset(LI.getPointerOperand(), LoadOff, DL);
265 if (!LoadBase) {
266 AnalysisFailed = true;
267 return;
268 }
269 LoadBase = LoadBase->stripPointerCasts();
270
271 SmallVector<Value *, 8> PtrWorkList;
272 SmallPtrSet<Value *, 16> LocalSeen;
273 PtrWorkList.push_back(LoadBase);
274
275 Value *CommonStoredVal = nullptr;
276 bool FoundStore = false;
277
278 while (!PtrWorkList.empty()) {
279 Value *Cur = PtrWorkList.pop_back_val();
280 if (!LocalSeen.insert(Cur).second)
281 continue;
282
283 for (User *U : Cur->users()) {
284 if (auto *SI = dyn_cast<StoreInst>(U)) {
285 int64_t StoreOff = 0;
286 Value *StoreBase = GetPointerBaseWithConstantOffset(
287 SI->getPointerOperand(), StoreOff, DL);
288 if (!StoreBase)
289 continue;
290 StoreBase = StoreBase->stripPointerCasts();
291
292 if (StoreBase != LoadBase || StoreOff != LoadOff)
293 continue;
294
295 Value *V = SI->getValueOperand();
296 if (!V->getType()->isPointerTy())
297 continue;
298
299 V = V->stripPointerCasts();
300 if (!FoundStore) {
301 CommonStoredVal = V;
302 FoundStore = true;
303 } else if (CommonStoredVal != V) {
304 AnalysisFailed = true;
305 return;
306 }
307 continue;
308 }
309
310 if (isa<GetElementPtrInst>(U) || isa<BitCastInst>(U) ||
311 isa<AddrSpaceCastInst>(U) || isa<PHINode>(U) ||
312 isa<SelectInst>(U)) {
313 PtrWorkList.push_back(cast<Value>(U));
314 continue;
315 }
316
317 if (auto *II = dyn_cast<IntrinsicInst>(U)) {
318 switch (II->getIntrinsicID()) {
319 case Intrinsic::dbg_declare:
320 case Intrinsic::dbg_value:
321 case Intrinsic::lifetime_start:
322 case Intrinsic::lifetime_end:
323 continue;
324 default:
325 break;
326 }
327 }
328 }
329 }
330
331 if (!FoundStore) {
332 AnalysisFailed = true;
333 return;
334 }
335
336 WorkList.push_back({CommonStoredVal, &LI});
337 }

358 void visitInsertValueInst(InsertValueInst &IVI) {
359 // Check if all inserted values derive from the same base ptr and match the
360 // aggregate field offsets they reconstruct.
361 auto IVIAggOffset =
362 getAggregateIndicesOffset(DL, IVI.getType(), IVI.getIndices());
363 if (!IVIAggOffset) {
364 AnalysisFailed = true;
365 AnalysisSuccess = false;
366 return;
367 }
368
369 int64_t IVIOffset = 0;
370 auto *BaseLoad = dyn_cast<LoadInst>(IVI.getInsertedValueOperand());
371 Value *BasePtrToCheck = BaseLoad ? BaseLoad->getPointerOperand()
372 : IVI.getInsertedValueOperand();
373 auto *InsertedValueBase =
374 GetPointerBaseWithConstantOffset(BasePtrToCheck, IVIOffset, DL);
375 if (!InsertedValueBase || IVIOffset != *IVIAggOffset) {
376 AnalysisFailed = true;
377 AnalysisSuccess = false;
378 return;
379 }
380
381 auto *IVINext = dyn_cast<InsertValueInst>(IVI.getAggregateOperand());
382 while (IVINext) {
383 auto NextAggOffset = getAggregateIndicesOffset(DL, IVINext->getType(),
384 IVINext->getIndices());
385 if (!NextAggOffset) {
386 AnalysisFailed = true;
387 AnalysisSuccess = false;
388 return;
389 }
390
391 int64_t NextOff = 0;
392 auto *Load = dyn_cast<LoadInst>(IVINext->getInsertedValueOperand());
393 Value *PtrToCheck =
394 Load ? Load->getPointerOperand() : IVINext->getInsertedValueOperand();
395 auto *NextAggregateBase =
396 GetPointerBaseWithConstantOffset(PtrToCheck, NextOff, DL);
397 if (NextAggregateBase != InsertedValueBase || NextOff != *NextAggOffset) {
398 // todo: are there IR examples where this actually matters?
399 AnalysisFailed = true;
400 AnalysisSuccess = false;
401 return;
402 }
403 // Go up the chain
404 IVINext = dyn_cast<InsertValueInst>(IVINext->getAggregateOperand());
405 }
406
407 DEBUG(Logger::logs("proteus-pass")
408 << "Insert value analysis found common base ptr : \n"
409 << *InsertedValueBase << "\n");
410
411 WorkList.push_back({InsertedValueBase, IVI.getInsertedValueOperand()});
412 }

474 void visitMemIntrinsic(MemIntrinsic &I) {
475 auto *MT = dyn_cast<MemTransferInst>(&I); // memcpy/memmove
476 if (!MT) {
477 // memset doesn't preserve any src->dst relationship we can use
478 AnalysisFailed = true;
479 return;
480 }
481
482 int64_t DstOff = 0, SrcOff = 0;
483 Value *DstBase =
484 GetPointerBaseWithConstantOffset(MT->getRawDest(), DstOff, DL);
485 Value *SrcBase =
486 GetPointerBaseWithConstantOffset(MT->getRawSource(), SrcOff, DL);
487 if (!DstBase || !SrcBase) {
488 AnalysisFailed = true;
489 return;
490 }
491
492 // Optional safety: only valid if the tracked byte lies within the copied
493 // region.
494 if (auto *LenC = dyn_cast<ConstantInt>(MT->getLength())) {
495 uint64_t Len = LenC->getZExtValue();
496 if (Offset < DstOff || uint64_t(Offset - DstOff) >= Len) {
497 AnalysisFailed = true;
498 return;
499 }
500 }
501
502 WorkList.push_back({SrcBase->stripPointerCasts(), &I});
503 Offset = Offset - DstOff + SrcOff;

605inline bool analyzeLambdaUses(
606 llvm::Module &M,
607 DenseMap<CallBase *, LambdaKernelArgAnalysis> &CallBaseToArgOffset,
608 const SmallVector<CallBase *> &CBToAnalyze) {
609 DEBUG(Logger::logs("proteus-pass") << "Beginning analysis " << "\n");
610 for (auto *FunctorCB : CBToAnalyze) {
611 LambdaArgVisitor Visitor(FunctorCB, M);
612 while (!Visitor.empty() && !Visitor.success() && !Visitor.failed()) {
613 auto [V, LastSeen] = Visitor.back();
614 Visitor.MemoryAnalysisPtrUse = LastSeen;
615 Visitor.popBack();
616 // Prevent loops/infinite recursion
617 if (Visitor.seen(V))
618 continue;
619 Visitor.markAsSeen(V);
620 DEBUG(Logger::logs("proteus-pass") << "Visiting value " << *V << "\n");
621 // Analyze the instruction
622 if (auto *I = dyn_cast<Instruction>(V))
623 Visitor.visit(*I);
624 else if (auto *A = dyn_cast<Argument>(V))
625 Visitor.visitArgument(*A);
626 else
627 continue;
628 }
629 if (!Visitor.success() || Visitor.failed()) {
630 DEBUG(Logger::logs("proteus-pass")
631 << "[WARNING]: Kernel arg analysis failed for functor beginning at "
632 << *FunctorCB << "\n");
633 return false;
634 }
635 LambdaKernelArgAnalysis Info = Visitor.getKernelArgInfo();
636 if (!Info.KernelFunction)
637 return false;
638 CallBaseToArgOffset[FunctorCB] = Info;
639 }
640 return true;
641}
