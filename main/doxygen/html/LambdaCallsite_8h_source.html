51 if (!Node || Node->getNumOperands() < 2)

56 auto *LambdaCI =

75inline LambdaKernelArgLocationVec getOrderedLambdaKernelArgLocations(

77 LambdaKernelArgLocationVec Ordered;

81 SmallVector<uint32_t, 8> Indices;
